            val = int(val)
            if val == 1:
                return "1 - Immediate"
            elif val == 2:
                return "2 - Emergent"
            elif val == 3:
                return "3 - Urgent"
            elif val == 4:
                return "4 - Less Urgent"
            elif val == 5:
                return "5 - Non-Urgent"
            else:
                return "3 - Urgent"
        except Exception:
            return "3 - Urgent"
